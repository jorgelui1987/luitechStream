Guía Rápida: ¿Dónde subir tus videos?

  • Opción Gratuita: Sube videos a Streamable, Vimeo, o CDN gratuitos y copia la URL directa .mp4
  • Con Laragon: Coloca tus videos .mp4 en C:\laragon\www\videos\ y usa la URL http://localhost/videos/tu-video.mp4
  • En Producción: Usa AWS S3, Cloudflare R2 o Firebase Storage y pega la URL pública

¿Necesita Base de Datos?

Esta versión (demo): NO necesita base de datos. Usa localStorage del navegador. Las series que subas se guardan localmente en tu PC.

Versión de producción: Recomendada con MySQL (Laragon) o Supabase/Firebase. Te dejo el script SQL listo abajo.

MySQL Schema (para producción con Laragon) 
-- Ejecutar en phpMyAdmin: http://localhost/phpmyadmin
CREATE DATABASE IF NOT EXISTS dramastream;
USE dramastream;

-- Tabla de series
CREATE TABLE IF NOT EXISTS series (
  id VARCHAR(50) PRIMARY KEY,
  title VARCHAR(200) NOT NULL,
  category VARCHAR(50) NOT NULL,
  cover VARCHAR(500) NOT NULL,
  description TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla de episodios
CREATE TABLE IF NOT EXISTS episodes (
  id INT AUTO_INCREMENT PRIMARY KEY,
  series_id VARCHAR(50) NOT NULL,
  ep_num INT NOT NULL,
  title VARCHAR(200) NOT NULL,
  video_url VARCHAR(500) NOT NULL,
  is_free BOOLEAN DEFAULT TRUE,
  cost INT DEFAULT 10,
  likes INT DEFAULT 0,
  FOREIGN KEY (series_id) REFERENCES series(id) ON DELETE CASCADE
);

-- Tabla de usuarios
CREATE TABLE IF NOT EXISTS users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  username VARCHAR(50) UNIQUE NOT NULL,
  coins INT DEFAULT 150,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla de desbloqueos
CREATE TABLE IF NOT EXISTS unlocks (
  user_id INT,
  series_id VARCHAR(50),
  ep_num INT,
  unlocked_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id, series_id, ep_num),
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

class DramaVideoPlayer extends StatefulWidget {
  final String url;
  final int episodeNum;
  const DramaVideoPlayer({super.key, required this.url, required this.episodeNum});

  @override
  State createState() => _DramaVideoPlayerState();
}

class _DramaVideoPlayerState extends State {
  late VideoPlayerController _controller;

  @override
  void initState() {
    super.initState();
    _controller = VideoPlayerController.networkUrl(Uri.parse(widget.url))
      ..initialize().then((_) {
        setState(() {});
        _controller.play();
        _controller.setLooping(true);
      });
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Stack(
      children: [
        _controller.value.isInitialized
            ? SizedBox.expand(
                child: FittedBox(
                  fit: BoxFit.cover,
                  child: SizedBox(
                    width: _controller.value.size.width,
                    height: _controller.value.size.height,
                    child: VideoPlayer(_controller),
                  ),
                ),
              )
            : const Center(child: CircularProgressIndicator(color: Colors.roseAccent)),
        Positioned(
          bottom: 40,
          left: 20,
          child: Text(
            "Capítulo ${widget.episodeNum}",
            style: const TextStyle(color: Colors.white, fontSize: 18, fontWeight: FontWeight.bold),
          ),
        )
      ],
    );
  }
}
Node.js Express Backend API (server.js) 
const express = require('express');
const cors = require('cors');
const mysql = require('mysql2/promise');
const app = express();

app.use(cors());
app.use(express.json());

// Configuración MySQL (Laragon)
const db = mysql.createPool({
  host: 'localhost',
  user: 'root',
  password: '',
  database: 'dramastream'
});

// Endpoint: Obtener Catálogo completo
app.get('/api/series', async (req, res) => {
  try {
    const [rows] = await db.query(
'SELECT s.*, e.ep_num, e.title as ep_title, e.video_url, e.is_free, e.cost ' +
      'FROM series s LEFT JOIN episodes e ON s.id = e.series_id'
    );
    // Agrupar episodios por serie
    const grouped = {};
    rows.forEach(r => {
      if (!grouped[r.id]) {
        grouped[r.id] = {
          id: r.id, title: r.title, category: r.category,
          cover: r.cover, description: r.description, episodes: []
        };
      }
      if (r.ep_num) {
        grouped[r.id].episodes.push({
          ep: r.ep_num, title: r.ep_title,
          url: r.video_url, isFree: !!r.is_free, cost: r.cost
        });
      }
    });
    res.json(Object.values(grouped));
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Endpoint: Desbloquear episodio con monedas
app.post('/api/unlock-episode', async (req, res) => {
  const { userId, seriesId, episodeNum } = req.body;
  try {
    // 1. Obtener costo del episodio
    const [epRows] = await db.query(
      'SELECT cost FROM episodes WHERE series_id = ? AND ep_num = ?',
      [seriesId, episodeNum]
    );
    if (epRows.length === 0) return res.status(404).json({ error: 'Episodio no encontrado' });
    const cost = epRows[0].cost;

    // 2. Verificar monedas del usuario
    const [userRows] = await db.query('SELECT coins FROM users WHERE id = ?', [userId]);
    if (userRows.length === 0) return res.status(404).json({ error: 'Usuario no encontrado' });
    if (userRows[0].coins < cost) return res.status(402).json({ error: 'Monedas insuficientes' });

    // 3. Descontar monedas y registrar desbloqueo
    await db.query('UPDATE users SET coins = coins - ? WHERE id = ?', [cost, userId]);
    await db.query(
      'INSERT INTO unlocks (user_id, series_id, ep_num) VALUES (?, ?, ?)',
      [userId, seriesId, episodeNum]
    );

    res.json({ success: true, message: 'Episodio desbloqueado con éxito', coinsLeft: userRows[0].coins - cost });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.listen(3000, () => console.log('Servidor corriendo en puerto 3000'));
